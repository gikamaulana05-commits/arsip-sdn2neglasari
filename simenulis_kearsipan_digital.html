Integrasi Apps Script (Code.gs & Google Drive)

Salin skrip backend ini untuk menghubungkan form upload langsung ke Google Drive & Sheets SDN 2 NEGLASARI

Langkah Singkat Pengaturan Google Drive & Sheets:
  1. Buat Folder Baru di Google Drive dengan nama: Arsip Surat SDN 2 Neglasari.
  2. Salin ID Folder dari URL (karakter acak setelah /folders/YOUR_FOLDER_ID).
  3. Buka Google Sheets baru & gt; menu Ekstensi > Apps Script, ganti isi Code.gs dengan kode di bawah.
  4. Ganti nilai variabel FOLDER_ID dengan ID folder Google Drive Anda.
  5. Klik Terapkan > Web App Baru (Set Akses: Siapa Saja / Anyone).
/* 
 * BACKEND GOOGLE APPS SCRIPT (Code.gs)
 * Kearsipan Digital - SDN 2 NEGLASARI (NPSN 20601631)
 */

// Ganti FOLDER_ID sesuai ID Folder Google Drive Sekolah Anda
var FOLDER_ID = "MASUKKAN_GOOGLE_DRIVE_FOLDER_ID_DISINI";

function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var jenis = data.jenis; // 'masuk' atau 'keluar'
    var sheetName = (jenis === 'masuk') ? "SuratMasuk" : "SuratKeluar";
    
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName(sheetName);
    
    if (!sheet) {
      sheet = ss.insertSheet(sheetName);
      if(jenis === 'masuk') {
        sheet.appendRow(["KodeKlasifikasi", "NoSurat", "Tanggal", "Pengirim", "Perihal", "FileURL", "Status"]);
      } else {
        sheet.appendRow(["KodeKlasifikasi", "NoSurat", "Tanggal", "Tujuan", "Perihal", "FileURL", "Status"]);
      }
    }
    
    var fileUrl = "";
    if (data.fileBase64 && data.fileName) {
      var folder = DriveApp.getFolderById(FOLDER_ID);
      var bytes = Utilities.base64Decode(data.fileBase64.split(',')[1] || data.fileBase64);
      var blob = Utilities.newBlob(bytes, data.fileMime || 'application/pdf', data.fileName);
      var file = folder.createFile(blob);
      file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
      fileUrl = file.getUrl();
    }
    
    sheet.appendRow([
      data.kode,
      data.noSurat,
      data.tanggal,
      data.pihak,
      data.perihal,
      fileUrl,
      jenis === 'masuk' ? "Menunggu Disposisi" : "Terkirim"
    ]);
    
    return ContentService.createTextOutput(JSON.stringify({
      result: "success",
      fileUrl: fileUrl
    })).setMimeType(ContentService.MimeType.JSON);
    
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({
      result: "error",
      message: error.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}
