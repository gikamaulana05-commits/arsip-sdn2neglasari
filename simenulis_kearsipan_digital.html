function doGet() {
  return HtmlService.createTemplateFromFile('index')
    .evaluate()
    .setTitle('Kearsipan Digital Sekolah')
    .addMetaTag('viewport', 'width=device-width, initial-scale=1.0')
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
}

function simpanSuratMasuk(formObject) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("SuratMasuk");
  if(!sheet) {
    sheet = SpreadsheetApp.getActiveSpreadsheet().insertSheet("SuratMasuk");
    sheet.appendRow(["KodeKlasifikasi", "NoSurat", "Tanggal", "Pengirim", "Perihal", "Status"]);
  }
  sheet.appendRow([
    formObject.kodeKlasifikasi,
    formObject.noSurat,
    formObject.tanggal,
    formObject.pengirim,
    formObject.perihal,
    "Belum Disposisi"
  ]);
  return { success: true };
}
